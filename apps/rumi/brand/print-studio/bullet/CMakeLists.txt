cmake_minimum_required(VERSION 3.20)
project(rumi_bullet LANGUAGES CXX)

if(NOT DEFINED BULLET_SOURCE_DIR OR NOT DEFINED NLOHMANN_INCLUDE_DIR)
  message(FATAL_ERROR "BULLET_SOURCE_DIR and NLOHMANN_INCLUDE_DIR must identify verified dependency sources")
endif()
set(USE_DOUBLE_PRECISION ON CACHE BOOL "" FORCE)
set(BUILD_BULLET3 OFF CACHE BOOL "" FORCE)
set(BUILD_BULLET2_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_CPU_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_OPENGL3_DEMOS OFF CACHE BOOL "" FORCE)
set(BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(INSTALL_LIBS OFF CACHE BOOL "" FORCE)
set(USE_GLUT OFF CACHE BOOL "" FORCE)
add_subdirectory("${BULLET_SOURCE_DIR}" bullet EXCLUDE_FROM_ALL)
add_executable(rumi-bullet main.cpp)
target_compile_features(rumi-bullet PRIVATE cxx_std_17)
target_compile_definitions(rumi-bullet PRIVATE BT_USE_DOUBLE_PRECISION)
target_include_directories(rumi-bullet PRIVATE "${BULLET_SOURCE_DIR}/src" "${NLOHMANN_INCLUDE_DIR}")
target_link_libraries(rumi-bullet PRIVATE BulletDynamics BulletCollision LinearMath)
set_target_properties(rumi-bullet PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(rumi-bullet PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
